// POST /api/admin/auth/register
router.post('/register', registerLimiter, async (req, res, next) => {
  try {
    // Check if admin already exists
    const hasAdmin = await adminExists()
    if (hasAdmin) {
      logAudit({
        action: 'REGISTER_FAILED',
        resource: 'User',
        req,
        success: false,
        metadata: { reason: 'Admin already exists', email: req.body.email }
      })
      
      res.status(403).json({ 
        error: 'Registration disabled. System already has an administrator.' 
      })
      return
    }
    
    // Validate input
    const validated = registerSchema.parse(req.body)
    
    // Create admin account
    const user = await register({
      email: validated.email,
      password: validated.password,
      role: 'ADMIN'
    })
    
    logAudit({
      userId: user.id,
      action: 'REGISTER',
      resource: 'User',
      resourceId: user.id,
      req,
      success: true,
      metadata: { email: user.email, role: 'ADMIN' }
    })
    
    res.status(201).json({
      success: true,
      user: { id: user.id, email: user.email, role: 'ADMIN' },
      message: 'Admin account created. You are the system administrator.'
    })
    
  } catch (error) {
    if (error instanceof z.ZodError) {
      logAudit({
        action: 'REGISTER_FAILED',
        resource: 'User',
        req,
        success: false,
        metadata: { error: error.errors[0].message, email: req.body.email }
      })
      
      res.status(400).json({ error: error.errors[0].message })
      return
    }
    if (error.message === 'Email already registered') {
      logAudit({
        action: 'REGISTER_FAILED',
        resource: 'User',
        req,
        success: false,
        metadata: { reason: 'Email already registered', email: req.body.email }
      })
      
      res.status(409).json({ error: 'Email already registered' })
      return
    }
    next(error)
  }
})

// POST /api/admin/auth/login
router.post('/login', loginLimiter, async (req, res, next) => {
  try {
    const validated = loginSchema.parse(req.body)
    
    const result = await login(
      { email: validated.email, password: validated.password },
      req
    )
    
    if (!result) {
      logAudit({
        action: 'LOGIN_FAILED',
        resource: 'User',
        req,
        success: false,
        metadata: { email: validated.email, reason: 'Invalid credentials' }
      })
      
      res.status(401).json({ error: 'Invalid credentials' })
      return
    }

    // Check if 2FA is enabled for this user
    const twoFAEnabled = await is2FAEnabled(result.user.id)
    
    // In login endpoint, when 2FA is required
if (twoFAEnabled) {
  const sessionId = randomBytes(16).toString('hex')
  const code = generate2FACode()
  await store2FACode(sessionId, code)
  
  // Store user data for later
  await redis.setex(
    `pending:${sessionId}`, 
    300, // 5 minutes
    JSON.stringify({
      userId: result.user.id,
      email: result.user.email,
      role: result.user.role
    })
  )
  
  console.log(`2FA Code for admin ${result.user.email}: ${code}`)
  
  res.json({
    requires2FA: true,
    sessionId
  })
  return
}
    
    // No 2FA - create session directly
    const token = generateToken({
      userId: result.user.id,
      email: result.user.email,
      role: result.user.role,
      sessionId: result.sessionId,
      tailscaleIP: (req as any).tailscaleIP
    })
    
    res.cookie('token', token, {
      httpOnly: true,
      secure: process.env.NODE_ENV === 'production',
      sameSite: 'strict',
      maxAge: 30 * 60 * 1000
    })
    
    logAudit({
      userId: result.user.id,
      action: 'LOGIN',
      resource: 'User',
      resourceId: result.user.id,
      req,
      success: true,
      metadata: { email: result.user.email, role: result.user.role }
    })
    
    res.json({
      success: true,
      user: result.user,
      token: token
    })
    
  } catch (error) {
    next(error)
  }
})

// POST /api/admin/auth/reset-password
router.post('/reset-password', resetLimiter, async (req, res, next) => {
  try {
    const { email, code, newPassword } = req.body;
    if (!email || !code || !newPassword) {
      return res.status(400).json({ error: 'Email, code, and new password required' });
    }

    // Retrieve from Redis
    const key = `reset:${email}:${code}`;
    const data = await redis.get(key);
    if (!data) {
      await logAudit({
      userId,
      action: 'RESET_PASSWORD-FAILED',
      resource: 'User',
      resourceId: userId,
      req,
      success: false,
      });
      return res.status(400).json({ error: 'Invalid or expired code' });
    }

    const { userId } = JSON.parse(data);
    
    // Verify user still exists and is admin
    const { data: user, error: userError } = await supabase
      .from('User')
      .select('id')
      .eq('id', userId)
      .eq('role', 'ADMIN')
      .single();
    
    if (userError || !user) {
      await redis.del(key);
      return res.status(400).json({ error: 'User no longer exists' });
    }
    // Hash new password
    const passwordHash = await hashPassword(newPassword);

    // Update user's password
    const { error: updateError } = await supabase
      .from('User')
      .update({ passwordHash })
      .eq('id', userId);

    if (updateError) throw updateError;

    // Delete the used code
    await redis.del(key);

    // Invalidate all sessions for this user (force re-login)
    await revokeAllUserTokens(userId);
    await logAudit({
      userId,
      action: 'RESET_PASSWORD',
      resource: 'User',
      resourceId: userId,
      req,
      success: true,
    });
    res.json({ success: true, message: 'Password updated successfully. Please log in again.' });
  } catch (error) {
    next(error);
  }
});

router.post('/verify-2fa', twoFALimiter, async (req, res, next) => {
  try {
    const validated = verify2FASchema.parse(req.body)
    
    const valid = await verify2FACode(validated.sessionId, validated.code)
    
    if (!valid) {
      // Log failed verification
      await logAudit({
        userId: undefined,
        action: 'VERIFY_2FA_FAILED',
        resource: 'User',
        req,
        success: false,
        errorMessage: 'Invalid or expired code',
        metadata: { sessionId: validated.sessionId }
      });
      
      res.status(401).json({ error: 'Invalid or expired code' })
      return
    }
    
    // Get the pending user data from Redis (store during login)
    const pendingUserData = await redis.get(`pending:${validated.sessionId}`)
    if (!pendingUserData) {
      throw new Error('Session expired or invalid')
    }
    
    const { userId, email, role } = JSON.parse(pendingUserData)
    
    // Create session
    const sessionId = randomBytes(16).toString('hex')
    const fingerprint = generateFingerprint(extractFromRequest(req))
    
    const { data: session, error: sessionError } = await supabase
      .from('Session')
      .insert({
        id: sessionId,
        userId,
        deviceFingerprint: fingerprint,
        ipAddress: req.ip,
        userAgent: req.headers['user-agent'],
        expiresAt: new Date(Date.now() + 30 * 60 * 1000).toISOString()
      })
      .select()
      .single()
    
    if (sessionError) throw sessionError
    
    // Generate access token (15 min)
    const accessToken = generateAccessToken({
      userId,
      email,
      role,
      sessionId,
      tailscaleIP: (req as any).tailscaleIP
    });

// Generate refresh token (7 days)
const refreshToken = await generateRefreshToken(userId)

// Set cookie (optional, can use localStorage instead)
res.cookie('token', accessToken, {
  httpOnly: true,
  secure: process.env.NODE_ENV === 'production',
  sameSite: 'strict',
  maxAge: 30 * 60 * 1000
});


// Log success
await logAudit({
  userId,
  action: 'VERIFY_2FA_SUCCESS',
  resource: 'User',
  resourceId: userId,
  req,
  success: true,
  metadata: { sessionId: validated.sessionId }
});
console.log(`[verify-2fa] Sending 200 response with refreshToken: ${refreshToken ? 'present' : 'missing'}`);

// Return BOTH tokens
res.json({
  success: true,
  token: accessToken,
  refreshToken: refreshToken,
  user: { id: userId, email, role }
});
    
  } catch (error) {
    next(error)
  }
})

        await revokeAllUserTokens(decoded.userId)
      }
    }
    
    res.clearCookie('token')
    res.json({ success: true })
    
  } catch (error) {
    next(error)
  }
})

// Helper to check if 2FA enabled
async function is2FAEnabled(userId: string): Promise<boolean> {
  try {
    const { data: user, error } = await supabase
      .from('User')
      .select('twoFactorEnabled')

      .from('User')
      .select('id')
      .eq('email', email)
      .single()
    
    let query = supabase
      .from('Booking')
      .select('*, event:Event(titleFr, titleAr, date)')
      .is('deletedAt', null)
      .order('createdAt', { ascending: false })
    
    // If user exists, filter by userId, otherwise return empty
    if (user) {
      query = query.eq('userId', user.id)
    } else {
      // Check for guest bookings by email (stored in notes or separate field)
      // For now, return empty
      res.json({ bookings: [] })
      return
    }
    
    const { data: bookings, error } = await query
    
    if (error) throw error
    
    res.json({ bookings })
    
  } catch (error) {
    next(error)
  }
})

    const guestEmail = req.body.guestEmail as string | undefined
    const guestName = req.body.guestName as string | undefined
    
    // Check event availability
    const { data: event, error: eventError } = await supabase
      .from('Event')
      .select('availableSlots, price')
      .eq('id', validated.eventId)
      .is('deletedAt', null)
      .single()
    
    if (eventError || !event) {
      res.status(404).json({ error: 'Event not found' })
      return
    }
    
    if (event.availableSlots < validated.childrenCount) {
      res.status(400).json({ error: 'Not enough available slots' })
      return
    }
    
    // Calculate total
    const totalPrice = event.price * validated.childrenCount
    
    // Try to find existing user by email
    let userId = null
    if (guestEmail) {
      const { data: user } = await supabase
        .from('User')
        .select('id')
        .eq('email', guestEmail)
        .single()
      userId = user?.id || null
    }
    
    // Encrypt notes if provided (store guest info)
    let notes = validated.notes || ''
    if (guestEmail && !userId) {
      // Store guest email in encrypted notes
      const guestInfo = JSON.stringify({ guestEmail, guestName })
      notes = notes ? `${notes} | ${guestInfo}` : guestInfo
    }
    
    const encryptedNotes = notes ? encrypt(notes) : null
    
    // Create booking
    const { data: booking, error } = await supabase
      .from('Booking')
      .insert({
        userId, // null for guest bookings
        eventId: validated.eventId,
        date: validated.date,
        childrenCount: validated.childrenCount,
        totalPrice,
        status: 'PENDING',
        paymentStatus: 'PENDING',
        notes: encryptedNotes
      })
      .select()
      .single()
    
    if (error) throw error
    
    // Decrement available slots
    await supabase
      .from('Event')
      .update({ availableSlots: event.availableSlots - validated.childrenCount })
      .eq('id', validated.eventId)
    
    res.status(201).json({ 
      booking,
      message: userId ? 'Booking confirmed' : 'Guest booking confirmed - please save your booking ID'
    })
    
  } catch (error) {
    if (error instanceof z.ZodError) {
      res.status(400).json({ error: error.errors[0].message })
      return
    }
    next(error)
  }
})

export default router

// Public product routes (Port 4000) - no auth required

import { Router } from 'express'
import { supabase } from '../../lib/supabase.js'

const router = Router()

// GET /api/public/products
router.get('/', async (req, res, next) => {
  try {
    const category = req.query.category as string | undefined
    
    let query = supabase
      .from('Product')
      .select('*')
      .is('deletedAt', null)
      .gt('stock', 0)
    
    if (category) {
      query = query.eq('category', category)
    }
    
    const { data: products, error } = await query.order('createdAt', { ascending: false })
    
    if (error) throw error
    
    res.json({ products })
    
  } catch (error) {
    next(error)
  }
})

// GET /api/public/products/:id
router.get('/:id', async (req, res, next) => {
  try {
    const { data: product, error } = await supabase
      .from('Product')
      .select('*')
      .eq('id', req.params.id)
      .is('deletedAt', null)
      .single()
    
    if (error || !product) {
      res.status(404).json({ error: 'Product not found' })
      return
    }
    
    res.json({ product })
    
  } catch (error) {
    next(error)
  }
})
