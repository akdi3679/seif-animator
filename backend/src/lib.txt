if (!req.cookies?.refreshToken) {
  await logoutUser(decoded.userId);

  return;
}
    const isRefreshEndpoint = req.path === '/auth/refresh';
    if (['POST', 'PUT', 'PATCH', 'DELETE'].includes(req.method) && !isRefreshEndpoint) {
      const csrfHeader = req.headers['x-csrf-token'] as string;
      if (!csrfHeader) {
        res.status(403).json({ error: 'CSRF token missing' });
        return;

      const a = Buffer.from(String(decoded.csrfToken));

}export async function validateRefreshToken(
  presentedToken: string
): Promise<RefreshValidationResult> {
  const familyId = await redis.get(tokenIndexKey(presentedToken));
  if (!familyId) return { kind: 'invalid' };

  const raw = await redis.get(familyKey(familyId));
  if (!raw) return { kind: 'invalid' };

  const record: RefreshFamilyRecord = JSON.parse(raw);

  // ---------- PASSWORD‑EPOCH CHECK ----------
 const { data: user } = await supabase
    .from('User')
    .select('passwordChangedAt')
    .eq('id', record.userId)
    .maybeSingle();
  if (!user || user.passwordChangedAt !== record.passwordUpdatedAt) {
    return { kind: 'password_changed', userId: record.userId, familyId: record.familyId };
  }

  // ---------- NORMAL VALIDATION ----------
  if (presentedToken === record.currentToken) {
    return { kind: 'fresh', userId: record.userId, familyId: record.familyId, passwordUpdatedAt: record.passwordUpdatedAt };
  }

  if (
    record.previousToken &&
    presentedToken === record.previousToken &&
    record.previousExpiresAt &&
    Date.now() < record.previousExpiresAt
  ) {
    return {
      kind: 'retry',
      userId: record.userId,
      familyId: record.familyId,
      reissueToken: record.currentToken,
      passwordUpdatedAt: record.passwordUpdatedAt,
    };
  }

  // Token found in family but not current / not in‑grace previous → replay
  return { kind: 'reused', userId: record.userId, familyId: record.familyId };

}export async function validateSession(sessionId: string, req: any): Promise<boolean> {
  try {
    let sessionData = await redis.get(`session:${sessionId}`);
    if (sessionData) {
      const session = JSON.parse(sessionData);
      if (new Date(session.expiresAt) < new Date()) {
        await redis.del(`session:${sessionId}`);
        return false;
      }
      const remainingMs = new Date(session.expiresAt).getTime() - Date.now();
      if (remainingMs < 10 * 60 * 1000) {
        const newExpiresAt = new Date(Date.now() + 30 * 60 * 1000).toISOString();
        session.expiresAt = newExpiresAt;
        await redis.setex(`session:${sessionId}`, 30 * 60, JSON.stringify(session));

        supabase.from('Session').update({ expiresAt: newExpiresAt }).eq('id', sessionId).then(({ error }) => {
          if (error) console.error('❌ Session DB extension failed:', error.message);
        });
      }

      return validateIntegrity(session, req);
    }

    // Supabase fallback
    const { data: session, error } = await supabase
      .from('Session')
      .select(`
        id,
        userId,
        deviceFingerprint,
        ipAddress,
        geoLocation,
        userAgent,
        expiresAt,
        user:User ( email, role )
      `)
      .eq('id', sessionId)
      .single();

   if (error && error.code !== 'PGRST116') {
  console.error('[validateSession] Supabase lookup failed:', error);
  return false; // fail closed — a sensitive check that couldn't be confirmed is treated as invalid
}
if (!session || new Date(session.expiresAt) < new Date()) {
  return false;
}
    const dataToCache = {
      userId: session.userId,
      email: (session.user as any).email,
      role: (session.user as any).role,
      deviceFingerprint: session.deviceFingerprint,
      ipAddress: session.ipAddress,
      geoLocation: session.geoLocation,
      userAgent: session.userAgent,
      expiresAt: session.expiresAt,
    };
    const remainingSecs = Math.floor((new Date(session.expiresAt).getTime() - Date.now()) / 1000);
    await redis
      .setex(`session:${sessionId}`, Math.max(remainingSecs, 1), JSON.stringify(dataToCache))
      .catch(err => console.error('❌ Redis repopulate failed:', err));

    return validateIntegrity(dataToCache, req);
  } catch (error) {
    console.error('Session validation error:', error);
    if (error instanceof Error && error.message === 'SESSION_CHECK_UNAVAILABLE') throw error;
    return false;
  }

async function manageConnectionWithBreaker(
  name: string,
  breaker: CircuitBreaker,
  checkFn: () => Promise<boolean>
): Promise<void> {
  const state = connections[name];
  
  if (state.connected) {
    // Silent health check via circuit breaker
    try {
      await breaker.fire();
      return;
    } catch {
      state.connected = false;
      state.failCount++;
      if (state.announced) {
        console.log(`⚠️  ${name} disconnected, circuit ${breaker.opened ? 'open' : breaker.pendingClose ? 'half-open' : 'closed'}`);
        state.announced = false;
      }
    }
  }
  
  // Try to reconnect using pRetry with exponential backoff
  try {
   await pRetry(
  async () => {
    try {
      const result = await breaker.fire();
      if (!result) throw new Error('Health check failed');
      return result;
    } catch (err) {
      // Log the actual error
      console.error(`⚠️  ${name} health check error:`, (err as Error).message || err);
      throw err;
    }
  },
  {
    retries: 5,
    factor: 2,
    minTimeout: 1000,
    maxTimeout: 30000,
    onFailedAttempt: (error) => {
      console.log(`⚠️  ${name} reconnect attempt ${error.attemptNumber} failed: ${(error as any).message}`);
    },
  }
);
    
    // Success
    state.connected = true;
    state.failCount = 0;
    if (!state.announced) {
      console.log(`✅ ${name} connected`);
      state.announced = true;
      if (name === 'supabase' && !adminCreated) {
        await createDefaultAdmin();
        adminCreated = true;
        console.log('✅ Admin creation check completed');
      }
    }
  } catch {
    state.connected = false;
    const delay = getRetryDelay(state.failCount);
   console.log(`⚠️  ${name} will retry in ${delay / 1000}s (circuit ${breaker.opened ? 'open' : breaker.pendingClose ? 'half-open' : 'closed'})`);
    setTimeout(() => manageConnectionWithBreaker(name, breaker, checkFn), delay);
  }
}

async function getTailscaleIP(): Promise<string> {
  if (process.env.TAILSCALE_SERVER_IP) {
    return process.env.TAILSCALE_SERVER_IP
  }
  try {
    const { stdout } = await execAsync('tailscale ip -4')
    return stdout.trim()
  } catch {
    if (process.env.NODE_ENV === 'production') {
      console.error('❌ Failed to get Tailscale IP in production. Set TAILSCALE_SERVER_IP env.')
      process.exit(1)
    }
    console.warn('⚠️ Tailscale not detected, falling back to localhost for development')
    return '127.0.0.1'
  }
}

// ============================================
// REQUEST SMUGGLING PROTECTION
// ============================================
function preventRequestSmuggling(req: express.Request, res: express.Response, next: express.NextFunction) {
  const hasTE = req.headers['transfer-encoding'] !== undefined
  const hasCL = req.headers['content-length'] !== undefined
  if (hasTE && hasCL) {
    return res.status(400).send('Bad Request: ambiguous headers')
  }
  next()
}

// ============================================
// WEB CACHE POISONING PROTECTION (strip client cache headers)
// ============================================
function stripCacheHeaders(req: express.Request, _res: express.Response, next: express.NextFunction) {
  delete req.headers['cache-control']
  delete req.headers['pragma']
  delete req.headers['expires']
  next()
}
