   1: /**
   2:  * Device Fingerprinting Middleware
   3:  * 
   4:  * Extracts and validates device fingerprint for session security
   5:  */
   6: 
   7: import { Request, Response, NextFunction } from 'express'
   8: import crypto from 'crypto'
   9: 
  10: // Extend Express Request type
  11: declare global {
  12:   namespace Express {
  13:     interface Request {
  14:       deviceFingerprint?: string
  15:     }
  16:   }
  17: }
  18: 
  19: /**
  20:  * Generate device fingerprint from request
  21:  */
  22: export function generateDeviceFingerprint(req: Request): string {
  23:   const components: string[] = []
  24: 
  25:   // User Agent
  26:   const userAgent = req.headers['user-agent'] || ''
  27:   components.push(`ua:${userAgent}`)
  28: 
  29:   // Accept Language
  30:   const acceptLanguage = req.headers['accept-language'] || ''
  31:   components.push(`lang:${acceptLanguage}`)
  32: 
  33:   // Accept Encoding
  34:   const acceptEncoding = req.headers['accept-encoding'] || ''
  35:   components.push(`enc:${acceptEncoding}`)
  36: 
  37:   // IP Address
  38:   const ip = req.ip || req.socket.remoteAddress || ''
  39:   components.push(`ip:${ip}`)
  40: 
  41:   // Create fingerprint hash
  42:   const fingerprint = crypto
  43:     .createHash('sha256')
  44:     .update(components.join('|'))
  45:     .digest('hex')
  46: 
  47:   return fingerprint
  48: }
  49: 
  50: /**
  51:  * Device fingerprint middleware
  52:  */
  53: export const deviceFingerprint = (req: Request, res: Response, next: NextFunction): void => {
  54:   const fingerprint = generateDeviceFingerprint(req)
  55:   req.deviceFingerprint = fingerprint
  56:   next()
  57: }
  58: 
  59: export default deviceFingerprint

   4: export async function validateTailscalePeer(
   5:   req: Request,
   6:   res: Response,
   7:   next: NextFunction
   8: ): Promise<void> {
   9:   try {
  10:     // 1. Validate the immediate source IP (frontend server)
  11:     const sourceIP = (req.headers['x-forwarded-for'] as string)?.split(',')[0].trim()
  12:                      || req.socket.remoteAddress
  13:                      || req.ip
  14:     const strippedSource = sourceIP.replace(/^::ffff:/, '')
  15: 
  16:     if (!strippedSource || !isTailscaleIP(strippedSource)) {
  17:       console.log('[Tailscale] Source IP not Tailscale:', strippedSource)
  18:       return res.status(404).json({ error: 'Not found' })
  19:     }
  20: 
  21:     const peers = await getTailscalePeers()
  22:     if (!peers.includes(strippedSource)) {
  23:       console.log('[Tailscale] Source IP not in peer list:', strippedSource)
  24:       return res.status(404).json({ error: 'Not found' })
  25:     }
  26: 
  27:     // 2. Validate the forwarded client IP (end user)
  28:     const clientIP = req.headers['x-forwarded-for'] as string
  29:     if (!clientIP) {
  30:       console.log('[Tailscale] Missing X-Forwarded-For header')
  31:       return res.status(404).json({ error: 'Not found' })
  32:     }
  33: 
  34:     const strippedClient = clientIP.split(',')[0].trim().replace(/^::ffff:/, '')
  35:     if (!strippedClient || !isTailscaleIP(strippedClient)) {
  36:       console.log('[Tailscale] Client IP not Tailscale:', strippedClient)
  37:       return res.status(404).json({ error: 'Not found' })
  38:     }
  39: 
  40:     if (!peers.includes(strippedClient)) {
  41:       console.log('[Tailscale] Client IP not in peer list:', strippedClient)
  42:       return res.status(404).json({ error: 'Not found' })
  43:     }
  44: 
  45:     // Both IPs are valid Tailscale peers
  46:     ;(req as any).tailscaleSourceIP = strippedSource
  47:     ;(req as any).tailscaleClientIP = strippedClient
  48:     next()
  49:   } catch (error) {
  50:     console.error('[Tailscale] Validation error:', error)
  51:     res.status(404).json({ error: 'Not found' })
  52:   }
  53: }
