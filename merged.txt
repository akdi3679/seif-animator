const RETRY_BASE_MS = 5000
const RETRY_MAX_MS = 60000

function getRetryDelay(failCount: number): number {
  return Math.min(RETRY_BASE_MS * Math.pow(2, failCount), RETRY_MAX_MS)
}

async function manageConnection(
  name: string,
  checkFn: () => Promise<boolean>,
  onConnect?: () => Promise<void>
) {
  const state = connections[name];

  // Try the health check directly (with pRetry for backoff)
  try {
    await pRetry(async () => {
      await checkFn();
    }, {
      retries: 3,
      factor: 2,
      minTimeout: 1000,
      maxTimeout: 10000,
      onFailedAttempt: (err) => {
        console.log(`⚠️  ${name} health check attempt ${err.attemptNumber} failed`);
      },
    });

    // Success
    if (!state.connected) {
      state.connected = true;
      state.failCount = 0;
      console.log(`✅ ${name} connected`);
      if (onConnect) await onConnect();
    }
    state.connected = true;
    state.failCount = 0;
  } catch {
    // All retries failed
    if (state.connected) {
      console.log(`⚠️  ${name} disconnected`);
      state.connected = false;
    }
    state.failCount++;
    const delay = getRetryDelay(state.failCount);
    console.log(`⚠️  ${name} will retry in ${delay / 1000}s`);
    setTimeout(() => manageConnection(name, checkFn, onConnect), delay);
  }
}

// REQUEST SMUGGLING PROTECTION
// ============================================
function preventRequestSmuggling(req: express.Request, res: express.Response, next: express.NextFunction) {
  const hasTE = req.headers['transfer-encoding'] !== undefined
  const hasCL = req.headers['content-length'] !== undefined
  if (hasTE && hasCL) {
    return res.status(400).send('Bad Request: ambiguous headers')
  }
  next()
}

// ============================================
// WEB CACHE POISONING PROTECTION (strip client cache headers)
// ============================================
function stripCacheHeaders(req: express.Request, _res: express.Response, next: express.NextFunction) {
  delete req.headers['cache-control']
  delete req.headers['pragma']
  delete req.headers['expires']
  next()
}

// ---------- Refresh token validation cache ----------
const refreshCache = new Map<string, { kind: string; expiresAt: number }>();
const CACHE_TTL_MS = 3_000; // 3 seconds micro‑cache
const MAX_CACHE_SIZE = 500;

async function validateRefreshCached(token: string) {
  const cached = refreshCache.get(token);
  if (cached && cached.expiresAt > Date.now()) return cached.kind;

  const result = await validateRefreshToken(token);
  const entry = { kind: result.kind, expiresAt: Date.now() + CACHE_TTL_MS };
  refreshCache.set(token, entry);

  // Simple LRU‑ish eviction: if too big, clear oldest half
  if (refreshCache.size > MAX_CACHE_SIZE) {
    const entries = [...refreshCache.entries()];
    entries.sort((a, b) => a[1].expiresAt - b[1].expiresAt);
    const toDelete = entries.slice(0, Math.floor(entries.length / 2));
    for (const [key] of toDelete) refreshCache.delete(key);
  }

  return result.kind;
}

// ============================================================================
// 3. csrfProtection — constant‑time CSRF header verification for mutations
// ============================================================================
export function csrfProtection(req: Request, res: Response, next: NextFunction): void {
  if (!['POST', 'PUT', 'PATCH', 'DELETE'].includes(req.method)) {
    return next();
  }

  const csrfHeader = req.headers['x-csrf-token'] as string;
  if (!csrfHeader) {
    clearAccessCookies(res);
    res.status(403).json({ error: 'CSRF token missing' });
    return;
  }

  const tokenCsrf = req.user?.csrfToken;
  if (!tokenCsrf) {
    clearAccessCookies(res);
    res.status(401).json({ error: 'Authentication required' });
    return;
  }

  const a = Buffer.from(String(tokenCsrf));
  const b = Buffer.from(String(csrfHeader));
  if (a.length !== b.length || !crypto.timingSafeEqual(a, b)) {
    clearAccessCookies(res);
    res.status(403).json({ error: 'Invalid CSRF token' });
    return;
  }

  next();
}

export async function validateTailscalePeer(
  req: Request,
  res: Response,
  next: NextFunction
): Promise<void> {
  try {
    // 1. Validate the immediate source IP (frontend server)
    const sourceIP = (req.headers['x-forwarded-for'] as string)?.split(',')[0].trim()

import { Request, Response, NextFunction } from 'express'
import redis from '../lib/redis.js'

interface RateLimitConfig {
  windowMs: number
  maxRequests: number
  keyPrefix: string
}

const DEFAULT_CONFIG: RateLimitConfig = {
  windowMs: 15 * 60 * 1000,
  maxRequests: 100,
  keyPrefix: 'ratelimit'
}

const ENDPOINT_LIMITS: Record<string, RateLimitConfig> = {
  '/api/public/auth/login': { windowMs: 15 * 60 * 1000, maxRequests: 5, keyPrefix: 'ratelimit:login' },
  '/api/public/auth/register': { windowMs: 60 * 60 * 1000, maxRequests: 3, keyPrefix: 'ratelimit:register' },
  '/api/public/auth/verify-2fa': { windowMs: 5 * 60 * 1000, maxRequests: 5, keyPrefix: 'ratelimit:2fa' },
  '/api/public/bookings': { windowMs: 60 * 60 * 1000, maxRequests: 10, keyPrefix: 'ratelimit:booking' },
  '/api/public/contact': { windowMs: 60 * 60 * 1000, maxRequests: 3, keyPrefix: 'ratelimit:contact' } , 
  '/api/public/booking-init': { windowMs: 60 * 60 * 1000, maxRequests: 10, keyPrefix: 'ratelimit:booking-init' },
  '/api/public/geo-lookup': { windowMs: 60 * 1000, maxRequests: 200, keyPrefix: 'ratelimit:geo' }
}

const AUTH_SENSITIVE_PREFIXES = ['login', 'register', '2fa', 'refresh', 'reset', 'forgot'];

export function createRateLimiter(config: Partial<RateLimitConfig> = {}) {
  return async (req: Request, res: Response, next: NextFunction): Promise<void> => {
    const finalConfig = { ...DEFAULT_CONFIG, ...config }
    const endpointConfig = ENDPOINT_LIMITS[req.path] || finalConfig
    const isAuthSensitive = AUTH_SENSITIVE_PREFIXES.some(p => endpointConfig.keyPrefix.includes(p));

    const ip = req.ip || req.headers['x-forwarded-for'] || 'unknown'
    const fingerprint = req.headers['x-device-fingerprint'] || 'unknown'
    const key = `${endpointConfig.keyPrefix}:${ip}:${fingerprint}`

    try {
      const ttl = Math.ceil(endpointConfig.windowMs / 1000)
      const count = await redis.incr(key)
      if (count === 1) { await redis.expire(key, ttl) }
      if (count > endpointConfig.maxRequests) {
        res.status(429).json({ error: 'Too many requests', retryAfter: ttl })
        return
      }
      res.setHeader('X-RateLimit-Limit', endpointConfig.maxRequests.toString())
      res.setHeader('X-RateLimit-Remaining', Math.max(0, endpointConfig.maxRequests - count).toString())
      next()
    } catch (err) {
      console.error(`[RateLimit] Redis unavailable for ${key}:`, err);
      if (isAuthSensitive) {
        res.status(503).json({ error: 'Service temporarily unavailable, please try again shortly' });
        return;
      }
      next();
    }
  }
}

import { Request, Response, NextFunction } from 'express';

// Routes that should always be allowed even under load (optional)
const CRITICAL_ROUTES = ['/health', '/api/public/health'];

export function systemLoadMiddleware(req: Request, res: Response, next: NextFunction) {
  // Skip health check endpoints
  if (CRITICAL_ROUTES.some(route => req.path.startsWith(route))) {
    return next();
  }

  if (!isSystemHealthy()) {
    const health = getSystemHealth();
    // Log for monitoring
    console.warn(`[SystemLoad] Rejecting request due to high load: CPU ${health.cpu.toFixed(1)}%, MEM ${health.memory.toFixed(1)}%, ELD ${health.eventLoopDelay.toFixed(2)}ms`);
    
    res.set('Retry-After', '30'); // seconds
    return res.status(503).json({
      error: 'Service temporarily unavailable due to high load',
      retryAfter: 30,
      reason: {
        cpu: health.cpu,
        memory: health.memory,
        eventLoopDelay: health.eventLoopDelay,
      },
    });
  }

  next();
}

import { supabase } from '../lib/supabase.js';
import { Request, Response, NextFunction } from 'express';
import crypto from 'crypto'; 
let cachedDeleteCode: string | null = null;
let cacheTime = 0;
const CACHE_TTL = 60_000;

async function getDeleteCode(): Promise<string | null> {
  const now = Date.now();
  if (cachedDeleteCode !== undefined && now - cacheTime < CACHE_TTL) {
    return cachedDeleteCode;
  }
  const { data } = await supabase
    .from('SecretCode')
    .select('code')
    .eq('purpose', 'lock_delete')
    .maybeSingle();
  cachedDeleteCode = data ? data.code : null;
  cacheTime = now;
  return cachedDeleteCode;
}

export async function verifyDeleteCode(req: Request, res: Response, next: NextFunction) {
  const { deleteCode } = req.body;
  const correct = await getDeleteCode();

  if (!deleteCode || typeof deleteCode !== 'string') {
    return res.status(400).json({ error: 'Delete verification required' });
  }
  if (!correct) {
    return res.status(400).json({ error: 'No delete verification code configured' });
  }
  const a = Buffer.from(String(deleteCode));
  const b = Buffer.from(String(correct));
  if (a.length !== b.length || !crypto.timingSafeEqual(a, b)) {
    return res.status(400).json({ error: 'Invalid delete code' });
  }
  next();
}

export async function validateSession(sessionId: string, req: any): Promise<boolean> {
  try {
    // 1. Try Redis
    const raw = await redis.get(`session:${sessionId}`);
    if (raw) {
      let session: any;
      try {
        session = JSON.parse(raw);
        console.log('[VALIDATE SESSION] id:', sessionId, 'source:', raw ? 'redis' : 'supabase (fallback)');
      } catch {
        // corrupted → treat as missing (will try Supabase)
        return await fallbackToSupabase(sessionId, req);
      }

      // Expired?
      if (new Date(session.expiresAt) < new Date()) {
        await redis.del(`session:${sessionId}`);
        return await fallbackToSupabase(sessionId, req);
      }

      // Extend TTL if less than 10 minutes remaining (fire-and-forget)
      const remainingMs = new Date(session.expiresAt).getTime() - Date.now();
      if (remainingMs < 10 * 60 * 1000) {
        const newExpiresAt = new Date(Date.now() + 30 * 60 * 1000).toISOString();
        session.expiresAt = newExpiresAt;
        redis.setex(`session:${sessionId}`, 30 * 60, JSON.stringify(session)).catch(() => {});
        supabase
          .from('Session')
          .update({ expiresAt: newExpiresAt })
          .eq('id', sessionId)
          .then(({ error }) => {
            if (error) console.error('Session extension DB update failed:', error.message);
          });
      }

      // Integrity check
      return validateIntegrity(session, req);
    }

    // 2. Redis miss – fallback to Supabase
    return await fallbackToSupabase(sessionId, req);
  } catch (error) {
    // Infrastructure errors (e.g. Redis timeout) should not crash the request
    console.error('Session validation error:', error);
    return false;
  }
}

      .from('Session')
      .select(`

        userId,
        deviceFingerprint,
        ipAddress,
        geoLocation,
        userAgent,

        user:User ( email, role )
      `)
      .eq('id', sessionId)

      return false;
    }
    if (!session || new Date(session.expiresAt) < new Date()) {
      return false;
    }

    const dataToCache = {

      role: (session.user as any).role,
      deviceFingerprint: session.deviceFingerprint,
      ipAddress: session.ipAddress,
      geoLocation: session.geoLocation,
      userAgent: session.userAgent,
      tailscaleIP: req.tailscaleClientIP,   // best-effort: store the current Tailscale IP for future checks
      expiresAt: session.expiresAt,
    };

    const remainingSecs = Math.floor((new Date(session.expiresAt).getTime() - Date.now()) / 1000);
    await redis
      .setex(`session:${sessionId}`, Math.max(remainingSecs, 1), JSON.stringify(dataToCache))

// ============================================================================
// token-service.ts — Grace‑period refresh token rotation (Redis‑based)
// with password‑epoch validation.
// ============================================================================

import { randomBytes } from 'crypto';
import jwt from 'jsonwebtoken';
import redis from '../lib/redis.js';
import { supabase } from '../lib/supabase.js';   // needed for password‑epoch check

const JWT_SECRET = process.env.JWT_SECRET!;
const ACCESS_TOKEN_EXPIRY = '30m';   // matches session TTL
const REFRESH_TOKEN_EXPIRY_DAYS = 7;
const REFRESH_TOKEN_TTL_SECONDS = REFRESH_TOKEN_EXPIRY_DAYS * 24 * 60 * 60;
const ROTATION_GRACE_SECONDS = 120;

export interface TokenPayload {
  userId: string;
  email: string;
  role: string;
  sessionId: string;
  tailscaleIP?: string;
  csrfToken: string;
  passwordEpoch?: string;            // user’s updatedAt at the moment of issuance
}

interface RefreshFamilyRecord {
  familyId: string;
  userId: string;
  currentToken: string;
  previousToken: string | null;
  previousExpiresAt: number | null;
  passwordUpdatedAt: string;          // user.updatedAt when this family was created
  issuedAccessTokenJti: string | null;
  createdAt: number;
   geoLocation?: any;  
}

function familyKey(familyId: string) {
  return `refresh_family:${familyId}`;
}

function tokenIndexKey(token: string) {
  return `refresh_token_idx:${token}`;
}

async function saveFamily(record: RefreshFamilyRecord) {
  const key = familyKey(record.familyId);
  await redis.set(key, JSON.stringify(record), 'EX', REFRESH_TOKEN_TTL_SECONDS);

  await redis.set(tokenIndexKey(record.currentToken), record.familyId, 'EX', REFRESH_TOKEN_TTL_SECONDS);
  if (record.previousToken) {
    await redis.set(tokenIndexKey(record.previousToken), record.familyId, 'EX', ROTATION_GRACE_SECONDS);
  }
}

/**
 * Generate a short‑lived access token.
 */
export function generateAccessToken(payload: TokenPayload): string {
  return jwt.sign(payload, JWT_SECRET, { expiresIn: ACCESS_TOKEN_EXPIRY });
}

/**
 * Start a brand‑new refresh‑token family. Call this on login / verify‑2fa.
 * `passwordUpdatedAt` must be the user’s current `updatedAt` value.
 */
export async function generateRefreshToken(
  userId: string,
  passwordUpdatedAt: string,
  geo?: any 
): Promise<string> {
  const familyId = randomBytes(16).toString('hex');
  const token = randomBytes(32).toString('hex');

  const record: RefreshFamilyRecord = {
    familyId,
    userId,
    currentToken: token,
    previousToken: null,
    previousExpiresAt: null,
    passwordUpdatedAt,
    issuedAccessTokenJti: null,
    createdAt: Date.now(),
      geoLocation: geo || null, 
  };

  await saveFamily(record);
  return token;
}

export type RefreshValidationResult =
  | { kind: 'fresh'; userId: string; familyId: string; passwordUpdatedAt: string; geoLocation?: any }
  | { kind: 'retry'; userId: string; familyId: string; reissueToken: string; passwordUpdatedAt: string; geoLocation?: any }
  | { kind: 'password_changed'; userId: string; familyId: string }
  | { kind: 'invalid' }
  | { kind: 'reused'; userId: string; familyId: string };

/**
 * Validate a presented refresh token.
 * Also checks the password epoch: if the password has changed since the
 * token was issued, the token is immediately rejected with `password_changed`.
 */
export async function validateRefreshToken(
  presentedToken: string
): Promise<RefreshValidationResult> {
  const familyId = await redis.get(tokenIndexKey(presentedToken));
  if (!familyId) return { kind: 'invalid' };

  const raw = await redis.get(familyKey(familyId));
  if (!raw) return { kind: 'invalid' };

  const record: RefreshFamilyRecord = JSON.parse(raw);

  // ---------- PASSWORD‑EPOCH CHECK (with transient error handling) ----------
  let user: { passwordChangedAt: string } | null = null;
  let lastError: any = null;

  // Retry once to survive a momentary Supabase blip
  for (let attempt = 0; attempt < 2; attempt++) {
    try {
      const result = await supabase
        .from('User')
        .select('passwordChangedAt')
        .eq('id', record.userId)
        .maybeSingle();
      if (result.error) throw result.error;
      user = result.data;
      break;   // succeeded – exit retry loop
    } catch (err) {
      lastError = err;
      if (attempt === 0) {
        // Brief wait before retry
        await new Promise(r => setTimeout(r, 200));
      }
    }
  }

  if (!user && lastError) {
    // Both attempts failed – this is an infrastructure problem, NOT a password change.
    // Throw a specific error that the /refresh handler can map to 503.
    throw new Error('SESSION_CHECK_UNAVAILABLE');
  }

  const dbEpoch = user?.passwordChangedAt
    ? new Date(user.passwordChangedAt).getTime().toString()
    : '0';
  const storedEpoch = record.passwordUpdatedAt;
  if (!user || dbEpoch !== storedEpoch) {
    return {
      kind: 'password_changed',
      userId: record.userId,
      familyId: record.familyId,
    };
  }

  // ---------- NORMAL VALIDATION ----------
  if (presentedToken === record.currentToken) {
    return {
      kind: 'fresh',
      userId: record.userId,
      familyId: record.familyId,
      passwordUpdatedAt: record.passwordUpdatedAt,
      geoLocation: record.geoLocation,
    };
  }

  if (
    record.previousToken &&
    presentedToken === record.previousToken &&
    record.previousExpiresAt &&
    Date.now() < record.previousExpiresAt
  ) {
    return {
      kind: 'retry',
      userId: record.userId,
      familyId: record.familyId,
      reissueToken: record.currentToken,
      passwordUpdatedAt: record.passwordUpdatedAt,
      geoLocation: record.geoLocation,
    };
  }

  // Token found in family but not current / not in‑grace previous → replay
  return { kind: 'reused', userId: record.userId, familyId: record.familyId };
}

/**
 * Rotate a refresh token after a *fresh* use.
 */
// Atomic compare-and-rotate: only rotates if the family record's
// currentToken still equals what THIS caller validated moments ago.
// Closes the double-tab race where two requests both read 'fresh'
// before either rotates — the second writer here is rejected instead
// of silently overwriting the first.
const ROTATE_SCRIPT = `
  local raw = redis.call('GET', KEYS[1])
  if not raw then return nil end
  local record = cjson.decode(raw)
  if record.currentToken ~= ARGV[1] then return false end
  record.previousToken = record.currentToken
  record.previousExpiresAt = tonumber(ARGV[2])
  record.currentToken = ARGV[3]
  redis.call('SET', KEYS[1], cjson.encode(record), 'EX', ARGV[4])
  redis.call('SET', KEYS[2], record.familyId, 'EX', ARGV[4])
  redis.call('SET', KEYS[3], record.familyId, 'EX', ARGV[5])
  return cjson.encode(record)
`;

export async function rotateRefreshToken(familyId: string, expectedCurrentToken: string): Promise<string | null> {
  const newToken = randomBytes(32).toString('hex');
  const result = await redis.eval(
    ROTATE_SCRIPT,
    3,
    familyKey(familyId),
    tokenIndexKey(newToken),
    tokenIndexKey(expectedCurrentToken),
    expectedCurrentToken,
    Date.now() + ROTATION_GRACE_SECONDS * 1000,
    newToken,
    REFRESH_TOKEN_TTL_SECONDS,
    ROTATION_GRACE_SECONDS
  );
  if (result === false || result === null) return null; // someone else already rotated this — caller must re-validate
  return newToken;
}

/**
 * Revoke an entire refresh family immediately.
 */
export async function revokeRefreshFamily(familyId: string): Promise<void> {
  const raw = await redis.get(familyKey(familyId));
  if (raw) {
    const record: RefreshFamilyRecord = JSON.parse(raw);
    await redis.del(tokenIndexKey(record.currentToken));
    if (record.previousToken) await redis.del(tokenIndexKey(record.previousToken));
  }
  await redis.del(familyKey(familyId));
}

/**
 * Revoke all refresh families for a user (logout‑all, password reset).
 */
export async function revokeAllUserTokens(userId: string): Promise<void> {
  let cursor = '0';
  do {
    const [next, keys] = await redis.scan(cursor, 'MATCH', 'refresh_family:*', 'COUNT', 100);
    cursor = next;
    for (const key of keys) {
      const raw = await redis.get(key);
      if (!raw) continue;
      const record: RefreshFamilyRecord = JSON.parse(raw);
      if (record.userId === userId) {
        await revokeRefreshFamily(record.familyId);
      }
    }
  } while (cursor !== '0');
}

  pixelRatio?: number
  touchSupport?: boolean
  canvas?: string
  webgl?: string
  fonts?: string[]
}

export function generateFingerprint(characteristics: DeviceCharacteristics): string {
  const components = [
    characteristics.userAgent,
    characteristics.acceptLanguage,
    characteristics.screenResolution || '',
    characteristics.timezone || '',
    characteristics.colorDepth?.toString() || '',
    characteristics.pixelRatio?.toString() || '',
    characteristics.touchSupport?.toString() || '',
    (characteristics.fonts || []).join(',')
  ];
  const data = components.join('|');
  const hash = createHash('sha256').update(data).digest('hex');
  console.log('[FINGERPRINT GENERATE] raw:', data);
  console.log('[FINGERPRINT GENERATE] hash:', hash);
  return hash;
}

// ----------------------------------------------------------------------
// Previously: any non-401/400 failure (503, network error, etc.) that
// survived 3 retry attempts fell through to clearAuthCookies() + redirect
// to /login?reason=session_expired — even though a 503 says NOTHING about
// whether the session/refresh token is valid. It only says the backend was
// briefly unavailable.
//
// Now: we explicitly distinguish two outcomes:
//   - AUTH-INVALID  (401/400 from the refresh endpoint itself): the token
//     really is bad. Clear cookies, force re-login. This is correct and
//     unchanged.
//   - TRANSIENT (503, network error, timeout, retries exhausted): we do
//     NOT touch cookies and we do NOT force a redirect. We surface a
//     distinguishable failure (ServerOverloadError) so callers — e.g.
//     SessionProvider — can decide to show "we're having trouble
//     connecting" UI and try again later, rather than logging the user out
//     of a perfectly valid session.
//

// /auth/refresh exempted from load-shedding) — together they mean a 503
// during refresh no longer has any path to deleting a good session.

export async function getDeviceFingerprint(): Promise<string> {
  if (cachedFingerprint) return cachedFingerprint;
  if (typeof window === 'undefined') return 'server-side';

  const components: string[] = [
    navigator.userAgent,
    navigator.language,
    screen.colorDepth.toString(),
    screen.width.toString(),
    screen.height.toString(),
    new Date().getTimezoneOffset().toString(),
    (navigator.hardwareConcurrency || 'unknown').toString(),
    (navigator.deviceMemory || 'unknown').toString(),
  ];

  try {
    const canvas = document.createElement('canvas');
    canvas.width = 200; canvas.height = 50;
    const ctx = canvas.getContext('2d');
    if (ctx) {
      ctx.textBaseline = 'top'; ctx.font = '14px Arial';
      ctx.fillStyle = '#f60'; ctx.fillRect(0, 0, 100, 50);
      ctx.fillStyle = '#069'; ctx.fillText('Myshow.tn', 2, 15);
      components.push(canvas.toDataURL().slice(0, 100));
    }
  } catch {}

  try {
    const gl = document.createElement('canvas').getContext('webgl') ||
               (document.createElement('canvas') as any).getContext('experimental-webgl');
    if (gl) {
      const debugInfo = gl.getExtension('WEBGL_debug_renderer_info');
      if (debugInfo) {
        components.push(gl.getParameter(debugInfo.UNMASKED_RENDERER_WEBGL));
        components.push(gl.getParameter(debugInfo.UNMASKED_VENDOR_WEBGL));
      }
    }
  } catch {}

  const encoder = new TextEncoder();
  const data = encoder.encode(components.join('|'));
  const hashBuffer = await crypto.subtle.digest('SHA-256', data);
  const hashArray = Array.from(new Uint8Array(hashBuffer));
  const hashHex = hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
  cachedFingerprint = hashHex.slice(0, 32);
  return cachedFingerprint;
}

async function request<T>(
  baseUrl: string,
  endpoint: string,
  options?: RequestInit,
  retry = true
): Promise<T> {
  const fingerprintHeaders: Record<string, string> = {};

  if (typeof window !== 'undefined') {
    fingerprintHeaders['X-Screen-Resolution'] = `${window.screen.width}x${window.screen.height}`;
    fingerprintHeaders['X-Timezone'] = Intl.DateTimeFormat().resolvedOptions().timeZone;
    fingerprintHeaders['X-Color-Depth'] = window.screen.colorDepth?.toString() || '';
    fingerprintHeaders['X-Pixel-Ratio'] = window.devicePixelRatio?.toString() || '';
    fingerprintHeaders['X-Touch-Support'] = ('ontouchstart' in window).toString();

    try {
      const canvas = document.createElement('canvas');
      canvas.width = 200; canvas.height = 50;
      const ctx = canvas.getContext('2d');
      if (ctx) {
        ctx.textBaseline = 'top'; ctx.font = '14px Arial';
        ctx.fillStyle = '#f60'; ctx.fillRect(0, 0, 100, 50);
        ctx.fillStyle = '#069'; ctx.fillText('Myshow.tn', 2, 15);
        fingerprintHeaders['X-Canvas-Fingerprint'] = canvas.toDataURL().slice(0, 100);
      }
    } catch {}

    try {
      const gl = document.createElement('canvas').getContext('webgl');
      if (gl) {
        const debugInfo = gl.getExtension('WEBGL_debug_renderer_info');
        if (debugInfo) {
          fingerprintHeaders['X-WebGL-Fingerprint'] = `${gl.getParameter(debugInfo.UNMASKED_RENDERER_WEBGL)}|${gl.getParameter(debugInfo.UNMASKED_VENDOR_WEBGL)}`;
        }
      }
    } catch {}
  }

  const fingerprint = await getDeviceFingerprint();
  const headers: Record<string, string> = {
    'Content-Type': 'application/json',
    'X-Device-Fingerprint': fingerprint,
    ...(options?.headers as Record<string, string>),
  };
  Object.assign(headers, fingerprintHeaders);

  const csrfToken = getCsrfTokenFromCookie();
  if (csrfToken && options?.method && !['GET', 'HEAD'].includes(options.method)) {
    headers['X-CSRF-Token'] = csrfToken;
  }
headers['X-Request-Timestamp'] = new Date()
  .toISOString()
  .replace('T', ' ')
  .replace(/\.\d{3}Z$/, ' +0000');
  const url = `${baseUrl}${endpoint}`;
  let response: Response;
  const isFormData = options?.body instanceof FormData;
  if (isFormData) {
    delete headers['Content-Type'];
  }
  try {
    response = await fetch(url, {
      ...options,
      headers,
      credentials: 'include',
    });
  } catch (fetchError: any) {
    const isConnectionRefused =
      fetchError?.message?.includes('ERR_CONNECTION_REFUSED') ||
      (fetchError?.name === 'TypeError' &&
        fetchError?.message === 'Failed to fetch' &&
        typeof navigator !== 'undefined' &&
        navigator.onLine);

    if (isConnectionRefused) {
      if (typeof window !== 'undefined') {
        window.dispatchEvent(new CustomEvent('backend-down'));
      }
      throw new ApiError('Service temporarily unavailable', 'BACKEND_DOWN', 0);
    }
    throw fetchError;
  }

  const AUTH_ENDPOINTS_NO_REFRESH = [
    '/auth/login',
    '/auth/verify-2fa',
    '/auth/resend-2fa',
    '/auth/forgot-password',
    '/auth/reset-password',
    '/auth/resend-reset-code',
    '/auth/reset-session-info',
    '/auth/2fa-session-info',
    '/auth/refresh',
    '/auth/status',
  ];

  if (response.status === 401 && retry) {
    if (AUTH_ENDPOINTS_NO_REFRESH.some(e => endpoint.includes(e))) {
      const data = await response.json().catch(() => ({}));
      throw new ApiError(data.error || 'Unauthorized', 'UNAUTHORIZED', 401);
    }

   try {
  const refreshed = await performRefresh();
  if (refreshed) {
    const freshCsrf = getCsrfTokenFromCookie();
    const retryOptions: RequestInit = { ...options };
    if (freshCsrf && options?.method && !['GET', 'HEAD'].includes(options.method)) {
      retryOptions.headers = {
        ...(options?.headers as Record<string, string>),
        'X-CSRF-Token': freshCsrf,
      };
    }
    // Retry the original request once
    try {
      return await request<T>(baseUrl, endpoint, retryOptions, false);
    } catch (retryErr) {
      // If the retry still fails with 401, the session is dead — force logout
      if (retryErr instanceof UnauthorizedError) {
        clearAuthCookies();
        redirectToLogin('session_expired');
        throw retryErr;
      }
      throw retryErr; // other errors (transient) propagate normally
    }
  }
} catch (error) {
  // performRefresh threw — it's already handled inside performRefresh (calls clearAuthCookies/redirect for real failures)
  throw error;
}
  }

  const contentType = response.headers.get('content-type');
  if (!contentType || !contentType.includes('application/json')) {
    const text = await response.text();
    throw new ApiError('Invalid response', 'INVALID_RESPONSE', response.status);
  }

  const data = await response.json();

  if (!response.ok) {
    if (response.status === 503) {
      if (typeof window !== 'undefined') window.dispatchEvent(new CustomEvent('backend-overload'));
      const retryAfter = response.headers.get('Retry-After') || '30';
      throw new ServerOverloadError('Server under high load', parseInt(retryAfter));
    }
    if (response.status === 429) {
      const retryAfter = parseInt(response.headers.get('Retry-After') || '60', 10);
      if (
        options?.method && ['GET', 'HEAD'].includes(options.method) &&
        baseUrl === PUBLIC_API_URL
      ) {
        window.dispatchEvent(new CustomEvent('backend-rate-limit'));
      }
      throw new RateLimitError(data.error || 'Rate limit exceeded', retryAfter);
    }
    throw new ApiError(data.error || 'Request failed', data.code, response.status);
  }

  return data as T;
}

// Generic fetch hook
export function useAPI<T>(
  fetcher: () => Promise<T>,
  deps: any[] = []
) {
  const [data, setData] = useState<T | null>(null)
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState<ApiError | null>(null)

  useEffect(() => {
    let mounted = true

    const fetchData = async () => {
      try {
        setLoading(true)
        const result = await withErrorHandling(fetcher(), { showToast: false })
        if (mounted) {
          setData(result)
          setError(null)
        }
      } catch (err) {
        if (mounted) {
          setError(err as ApiError)
        }
      } finally {
        if (mounted) {
          setLoading(false)
        }
      }
    }

    fetchData()

    return () => {
      mounted = false
    }
  }, deps)

  return { data, loading, error }
}

// Mutation hooks (POST, PUT, DELETE)
export function useMutation<T, D = any>(
  apiFunction: (data: D) => Promise<T>,
  options?: { showToast?: boolean; endpoint?: string }
) {
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<ApiError | null>(null);

  const mutate = useCallback(
    async (data: D): Promise<T> => {
      // Check client‑side rate limit
      if (options?.endpoint) {
        const cooldown = getRateLimitCooldown(options.endpoint);
        if (cooldown !== null) {
          const err = new RateLimitError('Too many requests', cooldown);
          setError(err);
          throw err;   // no backend call, immediate rejection
        }
      }

      setLoading(true);
      setError(null);
      try {
        const result = await withErrorHandling(apiFunction(data), {
          showToast: options?.showToast ?? true,
        });
        // Successful call → clear any cached limit for this endpoint
        if (options?.endpoint) clearRateLimit(options.endpoint);
        return result;
      } catch (err) {
        if (err instanceof RateLimitError && options?.endpoint) {
          // Cache the cooldown so future calls are blocked
          setRateLimit(options.endpoint, err.retryAfter);
        }
        setError(err as ApiError);
        throw err;
      } finally {
        setLoading(false);
      }
    },
    [apiFunction, options?.showToast]
  );

  return { trigger: mutate, mutate, loading, error };
}

export function usePublicGETSWR<T>(endpoint: string, options?: any) {
  const [retriesExhausted, setRetriesExhausted] = useState(false);
  const { data, error, mutate, isValidating } = useSWR<T>(endpoint, publicFetcher, {
    revalidateOnFocus: false,
    revalidateIfStale: false, // MAGIC BULLET: Prevents auto-refetch on page revisits
    dedupingInterval: 600000, // 10 minutes dedupe
    errorRetryCount: 3,
    keepPreviousData: false,
    revalidateOnMount: true,
    errorRetryInterval: 2000,
    onErrorRetry: (err, key, config, revalidate, { retryCount }) => {
      // Do NOT retry if backend is down, overloaded, or unauthorized
      if (
        (err as any)?.code === 'BACKEND_DOWN' ||
        (err as any)?.status === 0 ||
        err instanceof ServerOverloadError ||
        (err as any)?.status === 404 ||
        (err as any)?.status === 401 // STOPS 401 LOOP
      ) {
        setRetriesExhausted(true);
        return;
      }
      if (err instanceof RateLimitError) {
        setRetriesExhausted(true);
        return;
      }
      if (retryCount >= 3) {
        setRetriesExhausted(true);
        return;
      }
      if ((err as any)?.status >= 500) {
        if (retryCount < 1) {
          setTimeout(() => revalidate({ retryCount }), config.errorRetryInterval);
        } else {
          setRetriesExhausted(true);
        }
        return;
      }
      setTimeout(() => revalidate({ retryCount }), config.errorRetryInterval);
    },
    ...options,
  });

  const retry = useCallback(async () => {
    setRetriesExhausted(false);
    await mutate();
  }, [mutate]);

  return {
    data,
    loading: !data && !error,
    error,
    retry,
     mutate, 
    isRetrying: isValidating && !data,
    retriesExhausted,
    isValidating,  
  };
}

const TailscaleContext = createContext<{ healthy: boolean }>({ healthy: true });
export const useTailscaleHealth = () => useContext(TailscaleContext);

export function TailscaleGateProvider({ children }: { children: React.ReactNode }) {
  const pathname = usePathname();
  const router = useRouter();
  const [healthy, setHealthy] = useState(true);
  const checkedRef = useRef(false);

  const isProtectedPage =
    pathname?.startsWith('/admin') ||
    pathname?.includes('/login') ||
    pathname?.includes('/forgot-password') ||
    pathname?.includes('/reset-password') ||
    pathname?.includes('/verify-2fa');

  useEffect(() => {
    if (!isProtectedPage) return;

    const checkTailscale = async () => {
      try {
        const res = await fetch('/api/admin/health', { credentials: 'omit' });
        const ok = res.ok;
        setHealthy(ok);
        if (!ok) router.replace('/404');
      } catch {
        setHealthy(false);
        router.replace('/404');
      }
    };

    // Fetch only on the very first mount (strict‑mode guard)
    if (!checkedRef.current) {
      checkedRef.current = true;
      checkTailscale();
    }

    // Periodic recheck
    const interval = setInterval(checkTailscale, 2 * 60 * 1000);
    const onFocus = () => { if (document.visibilityState === 'visible') checkTailscale(); };
    document.addEventListener('visibilitychange', onFocus);

    return () => {
      clearInterval(interval);
      document.removeEventListener('visibilitychange', onFocus);
    };
  }, [isProtectedPage, pathname, router]);

  return (
    <TailscaleContext.Provider value={{ healthy }}>
      {children}
    </TailscaleContext.Provider>
  );
}

const GEO_CACHE_KEY = 'geo_data_v2';
const GEO_IP_KEY = 'geo_ip_v2';
const TTL = 30 * 60 * 1000;
   const PUBLIC_API_URL = process.env.NEXT_PUBLIC_API_URL || 'http://localhost:4000';

interface GeoData {
  country?: string;
  city?: string;
  lat?: number;
  lon?: number;
  timezone?: string;
}

let inMemoryGeo: GeoData | null = null;

export async function getCachedGeo(): Promise<GeoData | null> {
  if (inMemoryGeo) return inMemoryGeo;
  try {
    const cached = localStorage.getItem(GEO_CACHE_KEY);
    const cachedIP = localStorage.getItem(GEO_IP_KEY);
    if (cached) {
      const { data, timestamp, ip } = JSON.parse(cached);
      if (Date.now() - timestamp < TTL && ip === cachedIP) {
        inMemoryGeo = data;
        return data;
      }
    }
  } catch {}
  try {
const res = await fetch(`${PUBLIC_API_URL}/api/public/geo-lookup`);
    if (!res.ok) return null;
    const data = await res.json();
    const geo: GeoData = {
      country: data.country,
      city: data.city,
      lat: data.lat,
      lon: data.lon,
      timezone: data.timezone,
    };
    inMemoryGeo = geo;
    localStorage.setItem(GEO_CACHE_KEY, JSON.stringify({
      data: geo,
      timestamp: Date.now(),
      ip: data.ip,
    }));
    localStorage.setItem(GEO_IP_KEY, data.ip);
    return geo;
  } catch {
    return null;
  }
}

interface RateLimitEntry {
  expiresAt: number;   // timestamp when cooldown ends
}

const cache = new Map<string, RateLimitEntry>();

/**
 * Call this after receiving a RateLimitError.
 * Saves the endpoint + retryAfter so we can block subsequent calls.
 */
export function setRateLimit(endpoint: string, retryAfterSeconds: number) {
  cache.set(endpoint, {
    expiresAt: Date.now() + retryAfterSeconds * 1000,
  });
}

/**
 * Check if an endpoint is currently blocked.
 * Returns the remaining seconds if blocked, otherwise null.
 */
export function getRateLimitCooldown(endpoint: string): number | null {
  const entry = cache.get(endpoint);
  if (!entry) return null;
  const remaining = Math.ceil((entry.expiresAt - Date.now()) / 1000);
  return remaining > 0 ? remaining : null;
}

/**
 * Clear a cached rate limit (e.g., after a successful request).
 */
export function clearRateLimit(endpoint: string) {
  cache.delete(endpoint);
}

  logAudit({ userId: req.user.id, action: 'TEST_NOTIFICATION', resource: 'Notification', req, success: true });
  res.json({ success: true });
});
